+# Opt-in CompleteTech usage ledger
+
+Muse can export metadata-only usage through the public OpenAI-compatible provider
+fetch hook. CLI agent steps, compaction, memory extraction and the separate web
+chat route are instrumented. Collection is disabled by default; disabled mode
+returns the original fetch function and does not create files. The existing UI
+cost tracker remains unchanged and is not treated as a reported charge.
+
+The companion implementation and complete operator guide are in
+`CompleteTech-LLC/ai-usage-ledger-skill`, `references/muse-mesh.md`. Its Python
+standard-library MCP server uses Muse's existing `mcpServers` configuration;
+there is no additional Muse plugin framework or model provider dependency.
+
+Example `.mu.json` section, after reviewing the companion ledger's disclosure and
+consent requirements (API keys continue to use Muse's existing configuration):
+
+```json
+{
+  "usageLedger": {
+    "enabled": true,
+    "directory": "/absolute/private/mu-usage",
+    "sourceId": "workstation-mu",
+    "provider": "my-gateway",
+    "includeIdentifiers": false
+  }
+}
+```
+
+The source/provider label is operator supplied, not inferred from an endpoint or
+model name. The returned model and raw usage envelope are preferred to SDK zero
+fills. Missing counts remain null. The standard OpenAI-chat protocol has no
+separate cache-write count. `reportedChargeCurrency: "USD"` additionally permits
+reading an explicitly confirmed gateway `usage.cost` field; do not enable it
+without confirming that field's meaning. It is not an invoice or a list-price
+estimate. No prompt, completion, tool input/output, key, header, URL, request body
+or raw error is persisted. Existing Muse audit logs are outside this adapter and
+must be configured separately.
+
+Native response IDs are HMAC pseudonyms using an owner-only `.identity-key`, with
+persistent pre-request UUIDs as a fallback. `.source.json` prevents silently
+changing source/provider/privacy mode in a populated directory. Restore a lost
+key rather than regenerating it; stable identity is essential to replay safety.
+Each attempt writes a durable start marker and an atomic final file. A missing
+final is unfinished coverage, not zero usage. A bounded 256-write metadata queue
+retries temporary local write failures while allowing the model to continue;
+process termination before durable capture remains an explicit coverage gap.
+
+Collection, ledger import and MCP reporting are separate opt-ins. Giving usage
+reports to an MCP client may also give them to that client's remote model; local
+capture alone does not authorize this. The reporting server reads only approved
+source records; scan tools require additional explicit authorization and accept
+no path or configuration arguments. Read the companion guide before enabling
+personal data collection or reporting. POSIX directories/files must be 0700/0600;
+Windows operators must apply account-restricted ACLs.
+
+Run `pnpm test:usage-ledger` for native fixtures. Set `MU_LEDGER_REPO` to a local
+ledger checkout to enable the real Muse/SDK/SQLite/MCP end-to-end test; it is skipped
+without that environment variable. `PYTHON` can select the Python executable.
+`pnpm exec tsc --noEmit -p tsconfig.usage-ledger.json` checks production and test
+code. All verification uses deterministic local fixtures, not live providers.
+
+Disable with `usageLedger.enabled: false` or `MU_USAGE_LEDGER_DISABLED=true`.
+Remove the ledger MCP server entry and revoke reporting/import consent separately.
+Existing evidence and keys are not automatically deleted. This integration
+creates no schedule, background daemon or external data collector.

+/** Opt-in, metadata-only usage export at the public OpenAI-compatible fetch boundary. */
+import { AsyncLocalStorage } from 'node:async_hooks';
+import { createHmac, randomBytes, randomUUID } from 'node:crypto';
+import { closeSync, existsSync, fsyncSync, lstatSync, mkdirSync, openSync, readFileSync, readdirSync, renameSync, unlinkSync, writeFileSync } from 'node:fs';
+import { isAbsolute, join, resolve } from 'node:path';
+import type { MuConfig } from './config.js';
+
+export type UsageLedgerOptions = NonNullable<MuConfig['usageLedger']>;
+type Scope = { runId: string; operation: string; ordinal: number };
+type Json = Record<string, unknown>;
+const LIMIT = 1024 * 1024;
+const object = (value: unknown): Json => value !== null && typeof value === 'object' && !Array.isArray(value) ? value as Json : {};
+const text = (value: unknown): string | null => typeof value === 'string' && value.length > 0 && value.length <= 256 && !/[\x00-\x1f\x7f]/.test(value) ? value : null;
+const count = (value: unknown): number | null => typeof value === 'number' && Number.isSafeInteger(value) && value >= 0 ? value : null;
+
+/** No prompts, response text, URLs, headers or error messages enter this record. */
+export class UsageLedger {
+  private readonly scopes = new AsyncLocalStorage<Scope>();
+  private readonly pending = new Map<string, () => Json>();
+  private key?: Buffer;
+  private warned = false;
+  private readonly fallback: Scope = { runId: randomUUID(), operation: 'unscoped', ordinal: 0 };
+
+  constructor(private readonly options: UsageLedgerOptions | undefined, private readonly sessionId: string,
+    private readonly configuredModel: string, private readonly warn: (message: string) => void = () => {}) {}
+
+  run<T>(operation: 'agent' | 'memory' | 'web', fn: () => T): T {
+    if (!this.enabled()) return fn();
+    return this.scopes.run({ runId: randomUUID(), operation, ordinal: 0 }, fn);
+  }
+
+  private enabled(): boolean {
+    return this.options?.enabled === true && process.env.MU_USAGE_LEDGER_DISABLED !== 'true';
+  }
+
+  private warning(): void {
+    if (!this.warned) {
+      this.warned = true;
+      try { this.warn('Usage ledger export unavailable; agent operation continues. Check the local usage directory.'); } catch {}
+    }
+  }
+
+  private directory(): string {
+    const dir = this.options?.directory;
+    if (!dir || !isAbsolute(dir) || !this.options?.sourceId || !this.options?.provider) throw new Error('configuration');
+    // Refuse symlink components; require an owner-only leaf directory.
+    let current = resolve(dir);
+    while (current !== resolve(current, '..')) {
+      if (existsSync(current) && lstatSync(current).isSymbolicLink()) throw new Error('symlink');
+      current = resolve(current, '..');
+    }
+    mkdirSync(dir, { recursive: true, mode: 0o700 });
+    const st = lstatSync(dir);
+    if (!st.isDirectory() || (process.platform !== 'win32' && ((st.mode & 0o077) !== 0 || st.uid !== process.getuid?.()))) throw new Error('permissions');
+    const marker = { schema: 'mu-source/1', source_id: this.options.sourceId, provider: this.options.provider,
+      id_mode: this.options.includeIdentifiers === true ? 'literal' : 'hmac-sha256' };
+    const markerPath = join(dir, '.source.json');
+    try {
+      const fd = openSync(markerPath, 'wx', 0o600);
+      try { writeFileSync(fd, JSON.stringify(marker)); fsyncSync(fd); } finally { closeSync(fd); }
+    } catch (err) { if ((err as NodeJS.ErrnoException).code !== 'EEXIST') throw err; }
+    const markerStat = lstatSync(markerPath);
+    if (!markerStat.isFile() || markerStat.isSymbolicLink() || markerStat.size > 1024 ||
+      (process.platform !== 'win32' && ((markerStat.mode & 0o077) || markerStat.uid !== process.getuid?.()))) throw new Error('source marker');
+    const saved = JSON.parse(readFileSync(markerPath, 'utf8'));
+    if (Object.entries(marker).some(([key, value]) => saved[key] !== value)) throw new Error('source identity changed');
+    return dir;
+  }
+
+  private identity(value: string | null): string | null {
+    if (value === null) return null;
+    if (this.options?.includeIdentifiers === true) return value;
+    if (!this.key) {
+      const directory = this.directory();
+      const path = join(directory, '.identity-key');
+      if (!existsSync(path) && readdirSync(directory).some(name => /\.(usage|started)\.json$/.test(name))) {
+        throw new Error('restore the original identity key; never silently re-key an existing source');
+      }
+      try {
+        const fd = openSync(path, 'wx', 0o600);
+        try { writeFileSync(fd, randomBytes(32)); fsyncSync(fd); } finally { closeSync(fd); }
+      } catch (err) { if ((err as NodeJS.ErrnoException).code !== 'EEXIST') throw err; }
+      const stat = lstatSync(path);
+      if (!stat.isFile() || stat.isSymbolicLink() || stat.size !== 32 ||
+        (process.platform !== 'win32' && ((stat.mode & 0o077) || stat.uid !== process.getuid?.()))) throw new Error('key permissions or size');
+      const key = readFileSync(path);
+      if (key.length !== 32) throw new Error('key size');
+      this.key = key;
+    }
+    return createHmac('sha256', this.key).update(value).digest('hex');
+  }
+
+  private persist(name: string, record: Json): void {
+    const dir = this.directory();
+    const target = join(dir, name);
+    const temp = join(dir, `.${randomUUID()}.tmp`);
+    let fd: number | undefined;
+    try {
+      fd = openSync(temp, 'wx', 0o600);
+      writeFileSync(fd, JSON.stringify(record) + '\n');
+      fsyncSync(fd); closeSync(fd); fd = undefined;
+      renameSync(temp, target);
+      // POSIX directory fsync makes the rename durable. Windows does not support it.
+      if (process.platform !== 'win32') {
+        const dfd = openSync(dir, 'r');
+        try { fsyncSync(dfd); } finally { closeSync(dfd); }
+      }
+    } finally {
+      if (fd !== undefined) closeSync(fd);
+      try { unlinkSync(temp); } catch {}
+    }
+  }
+
+  /** Retry only metadata already captured in this process; never reissue a model request. */
+  flush(): void {
+    if (!this.enabled()) return;
+    for (const [name, record] of this.pending) {
+      try { this.persist(name, record()); this.pending.delete(name); this.warned = false; }
+      catch { this.warning(); break; }
+    }
+  }
+
+  private save(name: string, record: () => Json): void {
+    try { this.persist(name, record()); this.warned = false; }
+    catch {
+      // Bound memory. Start markers remain as coverage evidence if final persistence fails.
+      if (this.pending.size < 256 || this.pending.has(name)) this.pending.set(name, record);
+      this.warning();
+    }
+  }
+
+  /** Supported provider fetch injection; disabled mode returns the original function unchanged. */
+  wrapFetch(base: typeof globalThis.fetch = globalThis.fetch): typeof globalThis.fetch {
+    if (!this.enabled()) return base;
+    return async (url, init) => {
+      if (!this.enabled()) return base(url, init);
+      this.flush();
+      const scope = this.scopes.getStore() ?? this.fallback;
+      const id = randomUUID();
+      const ordinal = ++scope.ordinal;
+      const started = new Date().toISOString();
+      const common = (): Json => ({
+        schema: 'mu-usage/1', source_id: this.options!.sourceId, provider: this.options!.provider,
+        configured_model: this.configuredModel, invocation_id: id, session_id: this.identity(this.sessionId), run_id: this.identity(scope.runId),
+        operation: scope.operation, invocation_ordinal: ordinal, retry_of: null,
+        started_at: started, id_mode: this.options!.includeIdentifiers === true ? 'literal' : 'hmac-sha256',
+      });
+      this.save(`${id}.started.json`, () => ({ ...common(), status: 'started' }));
+      let model = this.configuredModel, responseId: string | null = null, timestamp: string | null = null;
+      let usage: Json = {}, completed = false, invalid = false, observed = false;
+      const capture = (data: unknown) => {
+        if (completed) return;
+        const d = object(data);
+        const nativeId = text(d.id);
+        if (nativeId) {
+          if (responseId !== null && responseId !== nativeId) invalid = true;
+          else responseId = nativeId;
+        }
+        model = text(d.model) ?? model;
+        if (typeof d.created === 'number' && Number.isFinite(d.created)) {
+          const date = new Date(d.created * 1000);
+          if (Number.isFinite(date.getTime())) timestamp = date.toISOString();
+        }
+        if (d.usage != null) { usage = object(d.usage); observed = true; }
+      };
+      const finish = (status: 'completed' | 'failed' | 'cancelled' | 'interrupted', httpStatus?: number) => {
+        if (completed) return;
+        completed = true;
+        const detail = object(usage.prompt_tokens_details);
+        const outputDetail = object(usage.completion_tokens_details);
+        for (const value of [usage.prompt_tokens, usage.completion_tokens, usage.total_tokens,
+          detail.cached_tokens, outputDetail.reasoning_tokens]) {
+          if (value != null && count(value) === null) invalid = true;
+        }
+        const observedResponseId = responseId;
+        const amount = usage.cost;
+        const charge = this.options?.reportedChargeCurrency === 'USD' && typeof amount === 'number' && Number.isFinite(amount) && amount >= 0 ? amount : null;
+        const record: Json = {
+          status, http_status: httpStatus ?? null, model,
+          timestamp: timestamp ?? started, timestamp_source: timestamp ? 'provider_created' : 'local_request_start',
+          usage_source: observed ? 'openai_chat_response_usage' : 'unavailable',
+          input_tokens: count(usage.prompt_tokens), output_tokens: count(usage.completion_tokens),
+          total_tokens: count(usage.total_tokens), cache_read_tokens: count(detail.cached_tokens),
+          cache_write_tokens: null, reasoning_tokens: count(outputDetail.reasoning_tokens),
+          input_semantics: 'openai-chat', capture_incomplete: invalid,

+      if (!response.body) { finish(response.ok ? 'completed' : 'failed', response.status); return response; }
